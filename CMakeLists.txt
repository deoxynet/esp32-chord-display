cmake_minimum_required(VERSION 3.16)
project(chord_display LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CHORD_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(CHORD_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB DETECTOR_SOURCES CONFIGURE_DEPENDS lib/ChordDetector/src/*.cpp)
list(FILTER DETECTOR_SOURCES EXCLUDE REGEX "(NoteUtils|ChordFormatter)\\.cpp$")
file(GLOB APP_SOURCES CONFIGURE_DEPENDS lib/ChordApp/src/*.cpp)
add_library(chord_core ${DETECTOR_SOURCES} ${APP_SOURCES})
target_include_directories(chord_core PUBLIC lib/ChordDetector/src lib/ChordApp/src)
target_compile_options(chord_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(chord_tests tests/test_main.cpp)
target_link_libraries(chord_tests PRIVATE chord_core)
add_test(NAME chord_tests COMMAND chord_tests)
add_executable(harmony_tests tests/test_harmony.cpp)
target_link_libraries(harmony_tests PRIVATE chord_core)
add_test(NAME harmony_tests COMMAND harmony_tests)
add_executable(jazz_theory_tests tests/test_jazz_theory.cpp)
target_link_libraries(jazz_theory_tests PRIVATE chord_core)
add_test(NAME jazz_theory_tests COMMAND jazz_theory_tests)
add_executable(candidate_tests tests/test_candidates.cpp)
target_link_libraries(candidate_tests PRIVATE chord_core)
add_test(NAME candidate_tests COMMAND candidate_tests)
add_executable(symbol_tests tests/test_symbols.cpp)
target_link_libraries(symbol_tests PRIVATE chord_core)
add_test(NAME symbol_tests COMMAND symbol_tests)
add_executable(chord_audit tools/chord_audit.cpp)
target_link_libraries(chord_audit PRIVATE chord_core)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME widi_protocol_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p test_widi_control.py)
  set_tests_properties(widi_protocol_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()

# Preview reads the installed device fonts; `pio pkg install -e core2` supplies
# them. Core tests do not depend on PlatformIO or display libraries.
set(M5GFX_FONTS "${CMAKE_SOURCE_DIR}/.pio/libdeps/core2/M5GFX/src/lgfx/Fonts" CACHE PATH "M5GFX font headers")
if(EXISTS "${M5GFX_FONTS}/GFXFF/FreeSansBold24pt7b.h")
  add_executable(chord_preview tools/preview.cpp)
  target_link_libraries(chord_preview PRIVATE chord_core)
  target_include_directories(chord_preview PRIVATE "${M5GFX_FONTS}")
endif()
